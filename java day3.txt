4. pyramid pattern

import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner (System.in);
        System.out.print("enter number :");
        int a = sc.nextInt();
        for (int i = 1; i <= a; i++) {
            for (int j = i; j < a; j++) {
                System.out.print(" ");
            }
            for (int k = 1; k <= (2 * i - 1); k++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }
}

for (int i = 1; i <= n; i++) {

for (int j = 1; j <= i; j++) {
System.out.print("*");
}

for (int j = 1; j <= 2 * (n - i); j++) {
System.out.print(" ");
